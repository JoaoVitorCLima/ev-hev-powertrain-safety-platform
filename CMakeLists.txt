cmake_minimum_required(VERSION 3.16)
project(ev_hev_safety_supervisor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ev_safety cpp/src/safety_supervisor.cpp)
target_include_directories(ev_safety PUBLIC cpp/include)
add_executable(safety_supervisor cpp/src/main.cpp)
target_link_libraries(safety_supervisor PRIVATE ev_safety)
if(MSVC)
  target_compile_options(ev_safety PRIVATE /W4)
  target_compile_options(safety_supervisor PRIVATE /W4)
else()
  target_compile_options(ev_safety PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(safety_supervisor PRIVATE -Wall -Wextra -Wpedantic)
endif()
